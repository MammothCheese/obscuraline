# =====================================================================================================================
#  OBSCURA LINE - THE LINE
# =====================================================================================================================
#
#  The stations and the legs between them. A travel chapter in puzzles.txt names a leg (leg = id); the title's map
#  draws every station at its pos and every leg as a line between its two stations (so a loop with branches is
#  just legs that share stations).
#
#  [station id]        lower case id
#      name   = AKARI HEIGHTS                    on screen: capitals, digits, spaces . ' -  (18 at most)
#      pos    = 22 50                        on the map: x y, 0..100 each (0 0 top left)
#      style  = metropolis                   its look: metropolis town industrial riverside onsen mvillage mountain
#                                            woodland rural sakura underground (the scenery round its platform on
#                                            every leg to or from it, its platform's length and canopy)
#      hidden = 1                            (optional) not on the map until every other station and line is done
#
#  [leg id]
#      line     = TSURU LOOP                 the line's name, in the cab's header (16 at most)
#      from     = shinmachi                  station ids (stations must come above)
#      to       = clockworks
#      dist     = 1100                       metres to the stop marker
#      time     = 110                        scheduled seconds (the autopilot's time + 2: python tests/retime.py sets it)
#      music    = 3                         (optional) the song (each its own key, tempo, beat, bass, pads, arp, swing):
#                                           0 neon departure, 1 night drive home, 2 rain run, 3 city funk, 4 outrun,
#                                           5 dub tunnel, 6 ascent, 7 the heart's, 8 the beach, 9 cherry house, 10 lighthouse
#      synth    = 480:950                   (optional) synth breaks (up to 3: 480:950 1400:1900; the limit holds 300 m
#                                           after each, to slow down from the cruise): metres where the line
#                                           runs at one limit (40+) and the keys play into the music (on the beat: up to
#                                           +5 each); the train cruises; no events in it
#      sky      = dusk                       day dusk night rain sunset dawn (rain: always; otherwise rain comes at
#                                           random, 15% of drives, never on sunset lines or ones mostly in tunnel, or with
#                                           rain = never (the first two lines);
#                                           wet rails: brakes 4/5, power 9/10, 6 s more on the timetable)
#      track    = single                     single (passing loops at stations) or double (trains on the other track)
#      platform = 75                         platform length, metres (40..200)
#      loop     = 1                          1: a passing loop at the arrival station (a train may wait in it)
#      limit    = 0:45 400:60 850:40         speed limits, km/h, from metre:value on (starts at 0)
#      curve    = 0:0 200:-20 380:0          curvature 12000/R: + right, - left (30 = 400 m radius)
#      grade    = 0:0 600:20 950:0           per mille, + uphill
#      zone     = 0:neon 250:houses          country town river tunnel street sea houses sakura neon beach festival
#      event    = 300:fans 850:check:40      railfans (horn), speed check board (SEL, km/h), oncoming:km/h (FX dim;
#                                            oncoming:0 = a train waiting, its front at that metre)
#      crossing = 150 420                    level crossings (metres)
#      ride     = tram                       (optional) the line is not driven but RIDDEN: a rhythm game to the song (music)
#                                           tram: the seaside tram (bells, REC at the crossing boards; slower, melodic)
#                                           festival: the festival tram (drums, lanterns; faster). A ride counts as driven
#                                           once passed; its tier is its score's (S at 95%). Marked apart on every map.
#      bars     = 40                         (a ride) bars of its song (40 at 90 BPM: about 1:50)
#      density  = 2                          (a ride) 1 sparse .. 3 busy
#      pass     = 0                          (a ride) % of the points to pass (0: the song played through is enough)
#
#  Every leg runs ONE WAY, from to to (a station can be left from the map once reached, so no way back is needed). Whistle boards (sound the
#  horn) are put before every tunnel and bridge on their own.
#
#  The network: the TSURU LOOP (seven stations), the SHORE BRANCH from Pacific to the coast (two), two RIDES (the
#  SEASIDE TRAM from Liberty to Crescent Bay, the FESTIVAL LINE from Union to Olympia), and a spur under Pacific to
#  Shinzo, the heart, that no map shows.
# =====================================================================================================================
